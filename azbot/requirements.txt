# ── Telegram MTProto core ────────────────────────────────────────────────
# pyrogram 2.x is the API this whole codebase is written against (HTML
# parse_mode on Client, filters.command, raw upload.SaveBigFilePart striped
# uploads). 2.0.106 is the final upstream release (project archived), so
# >=2.0.106 resolves to exactly it today; the <3 cap only guards against a
# future incompatible major rewrite silently breaking the bot.
pyrogram>=2.0.106,<3
# tgcrypto is NOT pinned here: 1.2.5 (last upstream release) has no cp312
# musllinux wheels and cannot build on python 3.14. The Dockerfile tries a
# best-effort `pip install tgcrypto` (source build) and continues without it
# on failure — crypto_guard.py self-tests at every startup and falls back to
# pure-Python crypto automatically. Worst case = slower crypto, never a crash.

# ── Downloaders (deliberately UNPINNED — near-daily fixes upstream) ──────
# NOT auto-updated at runtime (removed): `pip install -U yt-dlp` in the
# image / a bot restart is the update path.
yt-dlp
gallery-dl

# ── Support libs (stable public APIs; always current via pip) ────────────
requests
python-dotenv
google-api-python-client
google-auth
